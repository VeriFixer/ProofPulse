// Com - e contantes a linha fica protegida por assume, mas com y := x nao (talvez porque ja existe o check do x estar well formed?)
method Abs(x: int) returns (y: int)
  ensures x>=0 ==> 2*x==y
  ensures x<0 ==> 3*x==y
{
  if x < 0 {
    y := 3*x;
  } else {
    y := 2*x;
  }
}
