method CardinalitySubsetLt<T>(A: set<T>, B: set<T>)
  requires A < B
  ensures |A| < |B|
  decreases B
{
  var b :| b in B && b !in A;
  var B' := B - {b};
  assert |B| == |B'| + 1; // Assertion not essential for the prove
  if A < B' {
    CardinalitySubsetLt(A, B');
  } else {
    assert A == B'; // Essential Assertion for prove
  }
}
